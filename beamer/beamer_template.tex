\section{Mathematics}

\begin{frame}{Mathematical Content}
    \framesubtitle{Beautiful equations and theorem environments}
    
    The template supports all mathematical content with beamer-compatible theorem environments:
    
    \begin{align}
        \E[X] &= \sum_{i=1}^n x_i P(X = x_i) \\
        \Var(X) &= \E[X^2] - (\E[X])^2 \\
        \norm{\bm{x}}_2 &= \sqrt{\sum_{i=1}^n x_i^2}
    \end{align}
    
    \begin{definition}[Conditional Expectation]
        The conditional expectation of $X$ given $Y$ is:
        \[ \E[X \mid Y] = \sum_x x \cdot P(X = x \mid Y) \]
    \end{definition}
\end{frame}

\begin{frame}{More Mathematical Examples}
    \framesubtitle{Theorem environments in action}
    
    \begin{theorem}[Central Limit Theorem]
        Let $X_1, X_2, \ldots$ be i.i.d. random variables with $\E[X_i] = \mu$ and $\Var(X_i) = \sigma^2$. Then:
        \[ \frac{\sqrt{n}(\bar{X}_n - \mu)}{\sigma} \xrightarrow{d} N(0,1) \]
    \end{theorem}
    
    \begin{lemma}
        For any matrix $\bm{A} \in \R^{m \times n}$, we have $\norm{\bm{A}}_2 \leq \sqrt{mn} \norm{\bm{A}}_{\infty}$.
    \end{lemma}
    
    \begin{remark}
        These theorem environments maintain the color scheme from your trymtex.sty package while being fully beamer-compatible.
    \end{remark}
\end{frame}
